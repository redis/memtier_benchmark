# ThreadSanitizer suppressions for memtier_benchmark
#
# This file contains suppressions for known benign data races that do not
# affect correctness. These races are intentionally left unfixed for performance.

# Benign race on stats counters during progress display
# Worker threads update stats while main thread reads them for progress updates.
# This is benign because:
# - Progress stats are approximate and for display only
# - Final results are collected after pthread_join (race-free)
# - Adding synchronization would hurt performance
race:run_stats::set_end_time
race:run_stats::get_duration_usec
race:run_stats::get_total_ops
race:run_stats::get_total_bytes
race:run_stats::get_total_latency
race:run_stats::get_total_connection_errors
race:totals::update_op
race:totals::update_connection_error

# Additional benign-race read-side getters exercised by the Prometheus exporter's
# unconditional 1 Hz producer tick (same class as the display getters above).
race:run_stats::get_total_bytes_rx
race:run_stats::get_total_bytes_tx
race:run_stats::get_total_hits
race:run_stats::get_total_misses
race:run_stats::get_total_errors
race:run_stats::get_total_retry_attempts
race:run_stats::get_total_retried_ops

# Write-side inline counter bumps read by those getters.
race:run_stats::inc_error
race:run_stats::inc_retry_attempt
race:run_stats::inc_retried_op

# Inst-totals HDR histogram: workers record via hdr_record_value_capped_atomic with
# NO mutex (run_stats.cpp:44-52); the main thread copies under m_inst_histogram_mutex
# using plain reads (hdr_add / hdr_total_count inside run_stats::copy_inst_histogram and
# the gated copy_inst_histogram_if_changed, run_stats.cpp:197-202 + the Prometheus
# producer tick) — the mutex serializes copy-vs-reset only, so TSan reports
# atomic-write vs plain-read.
# Benign: live data is approximate; authoritative histograms merge race-free post-join.
race:run_stats::copy_inst_histogram
race:run_stats::copy_inst_histogram_if_changed

# OpenSSL internal races (false positives in libcrypto)
# These are known benign races within OpenSSL library itself
race:libcrypto.so*
